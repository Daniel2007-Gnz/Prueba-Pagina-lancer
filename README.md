# Prueba-Pagina-lancer

## 1. Crear el entorno virtual
python -m venv .env
## 2. Activar el entorno virtual entorno virtual
.\.env\Scripts\activate
## 3.Instalar el flask
python -m pip install 
## 4. Crear requierements.txt
pip freeze > requirements.txt
pip install -r requirements.txt
## 5. Ejecutar proyecto
python app.py
http://127.0.0.1:5000
## 6 Iniciar Git 
git init
git branch -M main
git branch
## 7 Concetar el proyecto con GitHub
git remote add origin https://github.com/Daniel2007-Gnz/Prueba-Pagina-lancer.git
